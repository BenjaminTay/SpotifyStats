"""Full yearly wrapped service — computes all 12 modules for the yearly review page."""

import sqlite3

import pandas as pd
import numpy as np

from backend.core.db import load_plays


def _hour(ms_series):
    return ms_series / 3_600_000


def _get_cover_url(conn: sqlite3.Connection, entity_type: str, name: str, artist_name: str = "") -> str:
    """Resolve cover URL for an entity. Returns empty string if not found."""
    if entity_type == "artist":
        row = conn.execute(
            "SELECT image_path, image_url FROM spotify_artist_meta WHERE artist_name = ? LIMIT 1",
            (name,),
        ).fetchone()
    elif entity_type == "album":
        row = conn.execute(
            "SELECT image_path, image_url FROM spotify_album_meta WHERE album_name = ? LIMIT 1",
            (name,),
        ).fetchone()
    elif entity_type == "track":
        row = conn.execute(
            "SELECT m.image_path, m.image_url FROM spotify_track_meta m "
            "JOIN tracks t ON t.spotify_track_uri = m.spotify_track_uri "
            "WHERE t.track_name = ? AND t.artist_name = ? LIMIT 1",
            (name, artist_name),
        ).fetchone()
    else:
        return ""
    if row:
        if row[0]:
            return f"/covers/{entity_type}s/{row[0]}"
        if row[1]:
            return row[1]
    return ""


def get_wrapped_full(
    conn: sqlite3.Connection, min_ms: int, music_only: bool, merge_enabled: bool, year: int,
) -> dict:
    """Generate complete yearly wrapped report with all 12 modules."""
    df = load_plays(conn, min_ms=min_ms, music_only=music_only, merge_enabled=merge_enabled)
    year_df = df[df["ts_year"] == year].copy()
    if year_df.empty:
        return {"year": year, "empty": True}

    # ═══ 1. Hero ═══
    total_minutes = year_df["ms_played"].sum() / 60_000
    total_plays = len(year_df)
    unique_tracks = year_df["track_id"].nunique()
    unique_artists = year_df["artist_name"].dropna().nunique()
    active_days = year_df["ts_date"].nunique()
    avg_minutes_per_day = total_minutes / active_days if active_days > 0 else 0
    avg_hours_per_day = total_minutes / 60 / max(active_days, 1)

    hero = {
        "total_minutes": round(total_minutes, 0),
        "total_plays": int(total_plays),
        "unique_tracks": int(unique_tracks),
        "unique_artists": int(unique_artists),
        "active_days": int(active_days),
        "avg_minutes_per_day": round(avg_minutes_per_day, 1),
    }

    # ═══ 2. Personality (7 dimensions) ═══
    # explorer
    unique_ratio = unique_tracks / max(total_plays, 1) * 100
    explorer_score = round(min(unique_ratio / 40 * 100, 100), 1)

    # loyalist
    top_artist_plays = year_df.groupby("artist_name").size().max()
    top_artist_share = top_artist_plays / max(total_plays, 1) * 100
    loyalist_score = round(min(top_artist_share / 20 * 100, 100), 1)

    # binger
    binger_score = round(min(avg_hours_per_day / 4 * 100, 100), 1)

    # night_owl (0-5am ratio)
    late_night_plays = len(year_df[year_df["ts_hour"].isin([0, 1, 2, 3, 4, 5])])
    night_owl_ratio = late_night_plays / max(total_plays, 1) * 100
    night_owl_score = round(min(night_owl_ratio / 50 * 100, 100), 1)

    # collector (album completion)
    # get album total track counts from meta
    album_names = year_df["album_name"].dropna().unique().tolist()
    collector_score = 0.0
    full_album_info = []
    if album_names:
        placeholders = ",".join(["?"] * len(album_names))
        meta_rows = conn.execute(
            f"SELECT album_name, total_tracks, track_list FROM spotify_album_meta WHERE album_name IN ({placeholders})",
            album_names,
        ).fetchall()
        album_meta = {r[0]: {"total_tracks": r[1], "track_list": r[2]} for r in meta_rows if r[1]}

        completed_albums = 0
        total_albums_with_meta = 0
        for album_name in album_names:
            am = album_meta.get(album_name)
            if not am or not am["total_tracks"]:
                continue
            total_albums_with_meta += 1
            album_tracks_played = year_df[year_df["album_name"] == album_name]["track_name"].nunique()
            completion_pct = album_tracks_played / am["total_tracks"] * 100
            if completion_pct >= 70:
                completed_albums += 1
        if total_albums_with_meta > 0:
            collector_score = round(completed_albums / total_albums_with_meta * 100, 1)

    # trend_chaser (new release ratio)
    # songs released in the target year vs total
    trend_chaser_score = 0.0
    track_names = year_df["track_name"].unique().tolist()
    if track_names:
        ph = ",".join(["?"] * len(track_names))
        release_rows = conn.execute(
            f"SELECT t.track_name, m.release_date FROM tracks t "
            f"JOIN spotify_track_meta m ON t.spotify_track_uri = m.spotify_track_uri "
            f"WHERE t.track_name IN ({ph})",
            track_names,
        ).fetchall()
        release_map = {}
        for r in release_rows:
            if r[1]:
                try:
                    rel_year = int(str(r[1])[:4])
                    release_map[r[0]] = rel_year
                except (ValueError, IndexError):
                    pass
        if release_map:
            new_count = sum(1 for t in year_df["track_name"] if release_map.get(t) == year)
            trend_chaser_score = round(min(new_count / max(total_plays, 1) * 100 / 50 * 100, 100), 1)

    # globetrotter (non-Chinese artist ratio)
    # use genres to determine Chinese vs non-Chinese
    artist_names = year_df["artist_name"].dropna().unique().tolist()
    globetrotter_score = 0.0
    if artist_names:
        ph = ",".join(["?"] * len(artist_names))
        genre_rows = conn.execute(
            f"SELECT artist_name, genres FROM spotify_artist_meta WHERE artist_name IN ({ph})",
            artist_names,
        ).fetchall()
        artist_genre_map = {}
        for r in genre_rows:
            genres_raw = r[1] or ""
            artist_genre_map[r[0]] = genres_raw.lower()
        chinese_keywords = ["mandopop", "cantopop", "c-pop", "chinese", "taiwan", "hokkien"]
        non_chinese_plays = 0
        total_with_genre = 0
        for name, count in year_df.groupby("artist_name").size().items():
            genres = artist_genre_map.get(name, "")
            if genres:
                total_with_genre += count
                if not any(kw in genres for kw in chinese_keywords):
                    non_chinese_plays += count
        if total_with_genre > 0:
            globetrotter_score = round(non_chinese_plays / total_with_genre * 100, 1)

    dimensions = {
        "explorer": {"label": "探索者", "score": explorer_score, "desc": "广泛涉猎不同曲目，保持音乐品味多样化"},
        "loyalist": {"label": "专一者", "score": loyalist_score, "desc": "对喜爱的艺人从一而终，深入了解他们的作品"},
        "binger": {"label": "狂听者", "score": binger_score, "desc": "音乐是日常必需品，每天大量时间沉浸在旋律中"},
        "night_owl": {"label": "夜猫子", "score": night_owl_score, "desc": "深夜是你和音乐独处的最好时光"},
        "collector": {"label": "收藏家", "score": collector_score, "desc": "你喜欢完整地听完一张专辑，而非单曲碎片"},
        "trend_chaser": {"label": "新潮派", "score": trend_chaser_score, "desc": "新歌上线总是第一时间尝鲜"},
        "globetrotter": {"label": "地球村", "score": globetrotter_score, "desc": "音乐无国界，你的播放列表跨越大洲大洋"},
    }

    # Determine primary personality
    priority_order = ["explorer", "binger", "night_owl", "trend_chaser", "collector", "globetrotter", "loyalist"]
    max_score = max(d["score"] for d in dimensions.values())
    primary_key = next((k for k in priority_order if dimensions[k]["score"] == max_score), "explorer")

    personality_map = {
        ("explorer", "globetrotter"): ("环球旅人", "你的播放列表就是一本没有签证的护照"),
        ("loyalist", "collector"): ("深度鉴赏家", "你不听歌，你研究歌。专辑是你的最小单位"),
        ("binger", "night_owl"): ("能量引擎", "音乐是你的日间燃料，从早燃到晚"),
        ("night_owl", "explorer"): ("午夜诗人", "凌晨两点，你和耳机里的世界清醒着"),
        ("trend_chaser", "explorer"): ("潮流捕手", "新歌上线，你是朋友圈第一个听到的人"),
        ("loyalist", "explorer"): ("忠实灯塔", "你知道自己要什么，一首歌可以循环一整年"),
    }

    # Select personality: find the best matching pair
    def _pick_personality(dims, p_key, p_map):
        # Single-dimension heuristic based on highest dimension
        top_two = sorted(dims.items(), key=lambda x: x[1]["score"], reverse=True)[:2]
        t1, t2 = top_two[0][0], top_two[1][0]
        pair = (t1, t2)
        if pair in p_map:
            return p_map[pair]
        # try reversed
        pair_rev = (t2, t1)
        if pair_rev in p_map:
            return p_map[pair_rev]
        # fallback: use primary key alone
        fallbacks = {
            "explorer": ("环球旅人", "你的播放列表就是一本没有签证的护照"),
            "loyalist": ("忠实灯塔", "你知道自己要什么，一首歌可以循环一整年"),
            "binger": ("能量引擎", "音乐是你的日间燃料，从早燃到晚"),
            "night_owl": ("午夜诗人", "凌晨两点，你和耳机里的世界清醒着"),
            "trend_chaser": ("潮流捕手", "新歌上线，你是朋友圈第一个听到的人"),
            "collector": ("深度鉴赏家", "你不听歌，你研究歌。专辑是你的最小单位"),
            "globetrotter": ("环球旅人", "你的播放列表就是一本没有签证的护照"),
        }
        return fallbacks.get(p_key, ("环球旅人", "音乐是你的世界护照"))

    primary_label, primary_desc = _pick_personality(dimensions, primary_key, personality_map)

    personality = {
        "primary": primary_key,
        "primary_label": primary_label,
        "primary_desc": primary_desc,
        "dimensions": dimensions,
    }

    # ═══ 3. Top Lists ═══
    top_artists_df = (
        year_df.groupby("artist_name")
        .agg(plays=("play_id", "count"), hours=("ms_played", _hour))
        .sort_values("hours", ascending=False)
        .head(5)
        .reset_index()
    )
    top_tracks_df = (
        year_df.groupby(["track_name", "artist_name"])
        .agg(plays=("play_id", "count"), hours=("ms_played", _hour))
        .sort_values("plays", ascending=False)
        .head(5)
        .reset_index()
    )
    top_albums_df = (
        year_df.groupby(["album_name", "artist_name"])
        .agg(hours=("ms_played", _hour))
        .sort_values("hours", ascending=False)
        .head(5)
        .reset_index()
    )

    top_lists = {
        "artists": [
            {
                "rank": i + 1,
                "name": r.artist_name or "",
                "plays": int(r.plays),
                "hours": round(float(r.hours), 1),
                "cover_url": _get_cover_url(conn, "artist", r.artist_name or ""),
            }
            for i, r in enumerate(top_artists_df.itertuples(index=False))
        ],
        "tracks": [
            {
                "rank": i + 1,
                "name": r.track_name or "",
                "artist_name": r.artist_name or "",
                "plays": int(r.plays),
                "hours": round(float(r.hours), 1),
                "cover_url": _get_cover_url(conn, "track", r.track_name or "", r.artist_name or ""),
            }
            for i, r in enumerate(top_tracks_df.itertuples(index=False))
        ],
        "albums": [
            {
                "rank": i + 1,
                "name": r.album_name or "",
                "artist_name": r.artist_name or "",
                "hours": round(float(r.hours), 1),
                "cover_url": _get_cover_url(conn, "album", r.album_name or ""),
            }
            for i, r in enumerate(top_albums_df.itertuples(index=False))
        ],
    }

    # ═══ 4. Genre Panorama ═══
    artist_names_for_genre = year_df["artist_name"].dropna().unique().tolist()
    genre_panorama = {"top_genres": [], "monthly_genres": [], "language_dist": {}}
    if artist_names_for_genre:
        ph = ",".join(["?"] * len(artist_names_for_genre))
        genre_rows = conn.execute(
            f"SELECT artist_name, genres FROM spotify_artist_meta WHERE artist_name IN ({ph}) AND genres IS NOT NULL AND genres != ''",
            artist_names_for_genre,
        ).fetchall()
        artist_genre_full = {}
        for r in genre_rows:
            artist_genre_full[r[0]] = [g.strip() for g in (r[1] or "").split(",") if g.strip()]

        if artist_genre_full:
            # Build genre → play_count mapping
            genre_plays = {}
            for artist, count in year_df.groupby("artist_name").size().items():
                genres = artist_genre_full.get(artist, [])
                for g in genres:
                    genre_plays[g] = genre_plays.get(g, 0) + count

            total_genre_plays = sum(genre_plays.values())
            sorted_genres = sorted(genre_plays.items(), key=lambda x: x[1], reverse=True)
            genre_panorama["top_genres"] = [
                {"name": g, "play_share": round(c / total_genre_plays * 100, 1)}
                for g, c in sorted_genres[:10]
            ]

            # Monthly genre evolution
            monthly_genres = []
            for month in range(1, 13):
                month_df = year_df[year_df["ts_month"] == month]
                month_genre_plays = {}
                for artist, count in month_df.groupby("artist_name").size().items():
                    genres = artist_genre_full.get(artist, [])
                    for g in genres:
                        month_genre_plays[g] = month_genre_plays.get(g, 0) + count
                month_total = sum(month_genre_plays.values())
                top5_month = sorted(month_genre_plays.items(), key=lambda x: x[1], reverse=True)[:5]
                monthly_genres.append({
                    "month": month,
                    "genres": {g: round(c / max(month_total, 1) * 100, 1) for g, c in top5_month},
                })
            genre_panorama["monthly_genres"] = monthly_genres

            # Language distribution (genre → language heuristic, delegated to frontend)

    # Daily grid: 12 rows (months) × 31 cols (days), value = play count

    daily_grid = []
    for m in range(1, 13):

        m_data = daily[daily["ts_month"] == m]

    monthly_pulse = (
        year_df.groupby("ts_month")

    )

    hourly_dist = (
        year_df.groupby("ts_hour")
        .size()
        .reset_index(name="plays")

        .size()

        .head(3)
        .reset_index(name="plays")
    )

    time_story = {
        "daily_grid": daily_grid,
        "monthly_pulse": [
            {"month": int(r.ts_month), "hours": round(float(r.hours), 1)}
            for r in monthly_pulse.itertuples(index=False)
        ],
        "hourly_dist": [
            {"hour": int(r.ts_hour), "plays": int(r.plays)}
            for r in hourly_dist.itertuples(index=False)
        ],

            "ratio": round(night_owl_ratio, 1),

                {
                    "name": r.track_name or "",
                    "artist_name": r.artist_name or "",
                    "plays": int(r.plays),
                    "cover_url": _get_cover_url(conn, "track", r.track_name or "", r.artist_name or ""),
                }
                for r in late_night_top.itertuples(index=False)
            ],

    # ═══ 6. Music Map ═══
    music_map = {"regions": [], "top_overseas_artists": []}
    if artist_genre_full:

    # New artists this year (first play for this artist in this year, never before)
    prev_df = df[df["ts_year"] < year]

    year_artists = set(year_df["artist_name"].dropna().unique())
    new_discovered = year_artists - prev_artists

    new_artist_plays = (

        .groupby("artist_name")
        .size()
        .sort_values(ascending=False)

        release_info = {}
        for r in release_rows:
            try:
                rel_year = int(str(r[2])[:4])
                release_info[(r[0], r[1])] = rel_year
            except (ValueError, IndexError):
                pass
        old_tracks = {
            (name, artist): rel_year
            for (name, artist), rel_year in release_info.items()
            if rel_year < year - 5
        }
        if old_tracks:
            old_track_plays = (
                year_df.groupby(["track_name", "artist_name"])

            old_track_plays = old_track_plays.sort_values("plays", ascending=False).head(5)
            discovery_returns["returning_tracks"] = [
                {
                    "name": r.track_name or "",
                    "artist_name": r.artist_name or "",
                    "plays": int(r.plays),
                    "release_year": old_tracks.get((r.track_name, r.artist_name), 0),
                    "cover_url": _get_cover_url(conn, "track", r.track_name or "", r.artist_name or ""),
                }
                for r in old_track_plays.itertuples(index=False)
            ]

    # Longest love (longest consecutive play span for a single track)
    year_df_sorted = year_df.sort_values("ts_date")
    longest_span = 0
    longest_track = None
    for (track_name, artist_name), track_df in year_df_sorted.groupby(["track_name", "artist_name"]):
        dates = sorted(track_df["ts_date"].unique())
        if len(dates) < 2:
            continue
        span = (pd.Timestamp(dates[-1]) - pd.Timestamp(dates[0])).days
        if span > longest_span:
            longest_span = span
            longest_track = (track_name, artist_name)
    if longest_track:
        discovery_returns["longest_love"] = {
            "name": longest_track[0] or "",
            "artist_name": longest_track[1] or "",
            "span_days": longest_span,
            "cover_url": _get_cover_url(conn, "track", longest_track[0] or "", longest_track[1] or ""),

    # Listening age

        ph = ",".join(["?"] * len(track_names))

        rel_years = []
        for r in release_rows:
            try:
                y = int(str(r[1])[:4])
                if 1900 < y <= year:
                    rel_years.append(y)
            except (ValueError, IndexError):
                pass
        if rel_years:
            avg_rel = int(sum(rel_years) / len(rel_years))
            music_age = year - avg_rel
            desc = ""
            if music_age < 5:
                desc = f"你的音乐品味紧跟时代，平均发行年份为 {avg_rel} 年，充满新鲜活力"
            elif music_age < 15:
                desc = f"你的歌单横跨经典与新潮，平均回溯到 {avg_rel} 年，品味均衡"
            elif music_age < 30:
                desc = f"你偏爱经典老歌，平均发行年份为 {avg_rel} 年，怀旧是你的底色"
            else:
                desc = f"你是一本行走的音乐史，最爱曲目平均发行于 {avg_rel} 年"
            listening_age = {
                "age": music_age,
                "avg_release_year": avg_rel,
                "description": desc,
            }

    # Album completion
    album_completion = []
    if album_names:
        for album_name in album_names:
            am = album_meta.get(album_name)
            if not am or not am["total_tracks"]:

            album_tracks_played = year_df[year_df["album_name"] == album_name]["track_name"].nunique()
            completion_pct = round(album_tracks_played / am["total_tracks"] * 100, 1)
            if completion_pct >= 70:
                artist = year_df[year_df["album_name"] == album_name]["artist_name"].iloc[0]
                album_completion.append({

                    "artist_name": artist or "",

                    "cover_url": _get_cover_url(conn, "album", album_name or ""),
                })
        album_completion = sorted(album_completion, key=lambda x: x["completion_pct"], reverse=True)[:3]

    deep_listen_ratio = 0.0
    if not year_df.empty:

        deep_count = len(year_df[year_df["ms_played"] >= 180000])
        deep_listen_ratio = round(deep_count / max(total_plays, 1) * 100, 1)

    listening_depth = {

        "album_completion": album_completion,

    most_active_day_track = ""
    most_active_day_cover = ""
    if most_active_date:
        top_track_day = (
            year_df[year_df["ts_date"] == most_active_date]
            .groupby(["track_name", "artist_name"]).size().sort_values(ascending=False)
        )
        if not top_track_day.empty:
            most_active_day_track = top_track_day.index[0][0]
            most_active_day_cover = _get_cover_url(conn, "track", top_track_day.index[0][0], top_track_day.index[0][1])

    hour_plays = year_df.groupby("ts_hour").size()
    earliest_hour = int(hour_plays.index.min()) if not hour_plays.empty else 0
    latest_hour = int(hour_plays.index.max()) if not hour_plays.empty else 0
    earliest_track_row = year_df[year_df["ts_hour"] == earliest_hour].iloc[0] if not year_df[year_df["ts_hour"] == earliest_hour].empty else None
    latest_track_row = year_df[year_df["ts_hour"] == latest_hour].iloc[0] if not year_df[year_df["ts_hour"] == latest_hour].empty else None

    sorted_dates = sorted(year_df["ts_date"].unique())
    longest_streak_val = 0
    streak_start = ""
    streak_end = ""
    current_streak = 1
    current_start = sorted_dates[0] if sorted_dates else ""
    for i in range(1, len(sorted_dates)):
        if (pd.Timestamp(sorted_dates[i]) - pd.Timestamp(sorted_dates[i-1])).days == 1:

        else:
            if current_streak > longest_streak_val:

                "artist_name": earliest_track_row["artist_name"] if earliest_track_row is not None else "",
                "cover_url": _get_cover_url(conn, "track",
                    earliest_track_row["track_name"] if earliest_track_row is not None else "",
                    earliest_track_row["artist_name"] if earliest_track_row is not None else ""),

        },
        "latest_listen": {
            "hour": latest_hour,
            "track": {

                "artist_name": latest_track_row["artist_name"] if latest_track_row is not None else "",

        "longest_streak": {
            "days": longest_streak_val,

    # ═══ 10. Monthly Drilldown ═══
    monthly_drilldown = []
    for month in range(1, 13):
        m_df = year_df[year_df["ts_month"] == month]
        m_hours = round(m_df["ms_played"].sum() / 3_600_000, 1)
        m_top_tracks = (
            m_df.groupby(["track_name", "artist_name"])
            .size().sort_values(ascending=False).head(3)

        comparison["last_year"] = {
            "total_hours_change": _pct_change(curr_total_hours, prev_total_hours),

export function inferLanguageDist(
  genres: { name: string; play_share: number }[]
): Record<string, number> {
  const dist: Record<string, number> = { chinese: 0, english: 0, korean: 0, japanese: 0, instrumental: 0, other: 0 }
  for (const g of genres) {
    const info = classifyGenre(g.name)
    dist[info.language] = (dist[info.language] || 0) + g.play_share
  }
  // Normalize to 100
  const total = Object.values(dist).reduce((a, b) => a + b, 0)
  if (total > 0) {
    for (const k of Object.keys(dist)) {
      dist[k] = Math.round(dist[k] / total * 1000) / 10
    }
  }
  return dist
}

export function inferRegionDist(
  genres: { name: string; play_share: number }[]
): { region: string; flag: string; play_share: number }[] {
  const regionMap: Record<string, { flag: string; share: number }> = {}
  for (const g of genres) {
    const info = classifyGenre(g.name)
    const key = info.region
    if (!regionMap[key]) regionMap[key] = { flag: info.flag, share: 0 }
    regionMap[key].share += g.play_share
  }
  return Object.entries(regionMap)
    .map(([region, v]) => ({ region, flag: v.flag, play_share: Math.round(v.share * 10) / 10 }))
    .sort((a, b) => b.play_share - a.play_share)
}

```typescript
import { useState, useEffect, useRef } from 'react'
import { get } from '@/lib/api'
import type { WrappedFullResponse } from '@/types/yearly-review'

// Module-level cache: year → data
const fullCache = new Map<number, WrappedFullResponse>()
// In-flight requests: year → Promise
const inFlight = new Map<number, Promise<WrappedFullResponse>>()

export function useYearlyReview(year: number) {
  const [data, setData] = useState<WrappedFullResponse | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)
  const mountedRef = useRef(true)

  useEffect(() => {
    mountedRef.current = true
    let cancelled = false

    async function load() {
      // Check cache
      const cached = fullCache.get(year)
      if (cached) {
        if (!cancelled) {
          setData(cached)
          setLoading(false)
        }
        return
      }

      // Deduplicate in-flight
      const existing = inFlight.get(year)
      if (existing) {
        try {
          const result = await existing
          if (!cancelled) {
            setData(result)
            setLoading(false)
          }
        } catch {
          // will be handled by the original request
        }
        return
      }

      setLoading(true)
      setError(null)

      const promise = get<WrappedFullResponse>(`/api/wrapped/${year}/full`)
      inFlight.set(year, promise)

      try {
        const result = await promise
        fullCache.set(year, result)
        if (!cancelled) {
          setData(result)
          setLoading(false)
        }
      } catch (err) {
        if (!cancelled) {
          setError(err instanceof Error ? err.message : 'Failed to load')
          setLoading(false)
        }
      } finally {
        inFlight.delete(year)
      }
    }

    load()

    return () => {
      cancelled = true
      mountedRef.current = false
    }
  }, [year])

  return { data, loading, error }
}

          你的年度音乐档案，用数据讲述这一年的听觉故事。
        </p>
      </section>

      {/* Year selector + Tabs */}
      <div className="flex items-center justify-between mb-8">
        <div className="flex gap-2">
          {AVAILABLE_YEARS.map(y => (
            <button
              key={y}
              onClick={() => setSearchParams({ year: String(y) })}
              className={cn(
                'px-4 py-1.5 rounded-full font-sans text-[13px] font-medium transition-colors',
                y === currentYear
                  ? 'bg-accent-foreground text-card'
                  : 'bg-muted text-muted-foreground hover:text-foreground'
              )}
            >
              {y}
            </button>
          ))}
        </div>
        <div className="flex gap-6 border-b border-border">
          {[{ key: 'custom', label: '年度总结' }, { key: 'official', label: '官方 Wrapped' }].map(tab => (
            <button
              key={tab.key}
              onClick={() => setActiveTab(tab.key as typeof activeTab)}
              className={cn(
                'pb-2.5 font-sans text-[13px] font-medium border-b-2 transition-colors -mb-[1px]',
                activeTab === tab.key
                  ? 'border-accent-foreground text-foreground font-semibold'
                  : 'border-transparent text-muted-foreground hover:text-foreground',
              )}
            >
              {tab.label}
            </button>
          ))}
        </div>
      </div>

      {/* Content */}
      {loading && <LoadingSkeleton />}
      {error && <ErrorState message={error} />}
      {data && !data.empty && activeTab === 'custom' && <CustomSummary data={data} />}
      {activeTab === 'official' && <OfficialWrapped />}
      {data?.empty && <EmptyState year={currentYear} />}
    </>
  )
}
```

## 注意事项

- **封面图**: 所有列出歌曲/艺人/专辑的地方必须展示 `cover_url`，fallback 为占位 SVG
- **详情跳转**: 使用 `<Link to={/music/artists/${encodeURIComponent(name)}}>` 等
- **空状态处理**: 每个模块在无数据时优雅降级（隐藏或显示提示），不要全页报错
- **年份范围**: 初始支持 2024-2025，后续可拓展
- **ECharts 动态加载**: 遵循项目现有模式 — 组件内 `import('echarts/...')`
- **深色模式**: 图表色盘使用 `getChartColors(isDark)` 跟随主题
